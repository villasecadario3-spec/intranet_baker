#!/usr/bin/env python3 """ Motor de extracción para escrituras de compraventa DS19/DS1/DS49. Recibe un PDF (escaneado o con texto), hace OCR página por página si hace falta, detecta las cláusulas numeradas, y busca dónde aparece cada uno de los 14 "requisitos" del formulario SERVIU, junto con el comprador (nombre, RUT), el programa de subsidio y el tramo. Uso: python3 extraer_escritura.py """ import sys import json import re import subprocess import tempfile import os # ── Números ordinales en español -> número, hasta trigésimo (30 cláusulas # debería ser más que suficiente para este tipo de escritura). ORDINALES = { 'PRIMERO': 1, 'SEGUNDO': 2, 'TERCERO': 3, 'CUARTO': 4, 'QUINTO': 5, 'SEXTO': 6, 'SEPTIMO': 7, 'SÉPTIMO': 7, 'OCTAVO': 8, 'NOVENO': 9, 'DECIMO': 10, 'DÉCIMO': 10, 'DECIMO PRIMERO': 11, 'DÉCIMO PRIMERO': 11, 'UNDECIMO': 11, 'UNDÉCIMO': 11, 'DECIMO SEGUNDO': 12, 'DÉCIMO SEGUNDO': 12, 'DUODECIMO': 12, 'DUODÉCIMO': 12, 'DECIMO TERCERO': 13, 'DÉCIMO TERCERO': 13, 'DECIMO CUARTO': 14, 'DÉCIMO CUARTO': 14, 'DECIMO QUINTO': 15, 'DÉCIMO QUINTO': 15, 'DECIMO SEXTO': 16, 'DÉCIMO SEXTO': 16, 'DECIMO SEPTIMO': 17, 'DÉCIMO SÉPTIMO': 17, 'DECIMO OCTAVO': 18, 'DÉCIMO OCTAVO': 18, 'DECIMO NOVENO': 19, 'DÉCIMO NOVENO': 19, 'VIGESIMO': 20, 'VIGÉSIMO': 20, 'VIGESIMO PRIMERO': 21, 'VIGÉSIMO PRIMERO': 21, 'VIGESIMO SEGUNDO': 22, 'VIGÉSIMO SEGUNDO': 22, 'VIGESIMO TERCERO': 23, 'VIGÉSIMO TERCERO': 23, 'VIGESIMO CUARTO': 24, 'VIGÉSIMO CUARTO': 24, 'VIGESIMO QUINTO': 25, 'VIGÉSIMO QUINTO': 25, 'VIGESIMO SEXTO': 26, 'VIGÉSIMO SEXTO': 26, 'VIGESIMO SEPTIMO': 27, 'VIGÉSIMO SÉPTIMO': 27, 'VIGESIMO OCTAVO': 28, 'VIGÉSIMO OCTAVO': 28, 'VIGESIMO NOVENO': 29, 'VIGÉSIMO NOVENO': 29, 'TRIGESIMO': 30, 'TRIGÉSIMO': 30, 'CUADRAGESIMO': 40, 'CUADRAGÉSIMO': 40, } # Se ordenan de más largo a más corto para que el regex intente primero # "DECIMO SEGUNDO" antes que "DECIMO" solo (si no, matchearía el más corto). ORDINALES_ORDENADOS = sorted(ORDINALES.keys(), key=len, reverse=True) PATRON_CLAUSULA = re.compile( r'^(' + '|'.join(re.escape(o) for o in ORDINALES_ORDENADOS) + r')\s*:', re.MULTILINE ) # ── Diccionario de requisitos -> palabras clave a buscar (basado en el # lenguaje real observado en escrituras DS19 de Baker). Si no encuentra # ninguna, el campo queda vacío para que la persona lo revise a mano. REQUISITOS = { 'Permiso Edificación': [r'permiso de edificaci'], 'Recepción Final': [r'recepci[oó]n final', r'recepci[oó]n definitiva'], 'N° y fecha Resolución': [r'resoluci[oó]n exenta', r'resoluci[oó]n n[uú]mero'], 'Estacionamiento': [r'estacionamiento'], 'Individualización vivienda': [r'antecedentes', r'inmueble'], # cae en la cláusula que describe la propiedad 'Precio compraventa': [r'precio de venta', r'precio de.*compraventa'], 'Seguro incendio y sismo': [r'seguro.*incendio', r'incendio y sismo'], 'Seguro desgravamen e invalidez 2/3': [r'seguro de desgravamen', r'invalidez permanente'], 'Seguro desempleo e incapacidad temporal': [r'seguro de desempleo', r'incapacidad temporal'], 'Prohibiciones de enajenar, ceder uso y goce y habitar en favor de Serviu': [r'prohibici[oó]n de enajenar'], 'Mandato cobro subsidio': [r'mandato[\s\S]{0,150}?cobr\w*[\s\S]{0,60}?subsidio', r'subsidio[\s\S]{0,150}?mandato'], 'Exención IVA o Contrato suma alzada': [r'exent[oa].*iva', r'suma alzada'], 'Nómina de Profesionales, articulo 18. Ley General de U.': [r'n[oó]mina de profesionales', r'art[ií]culo\s*18'], } PATRON_PROGRAMA = { 'DS19': [r'D\.?\s*S\.?\s*N[°ºo]?\s*19', r'decreto supremo n[uú]mero diecinueve'], 'DS1': [r'D\.?\s*S\.?\s*N[°ºo]?\s*1\b', r'decreto supremo n[uú]mero uno\b'], 'DS49': [r'D\.?\s*S\.?\s*N[°ºo]?\s*49', r'decreto supremo n[uú]mero cuarenta y nueve'], } PATRON_RUT = re.compile(r'\b(\d{1,2}\.?\d{3}\.?\d{3}[-‐]?[\dkK])\b') def ocr_pdf(ruta_pdf, dpi=300): """OCR página por página. Devuelve lista de (num_pagina, texto).""" with tempfile.TemporaryDirectory() as tmp: prefijo = os.path.join(tmp, 'pag') subprocess.run( ['pdftoppm', '-gray', '-r', str(dpi), ruta_pdf, prefijo], check=True, capture_output=True ) archivos = sorted(f for f in os.listdir(tmp) if f.endswith('.pgm')) resultado = [] for i, nombre in enumerate(archivos, start=1): ruta_img = os.path.join(tmp, nombre) texto = subprocess.run( ['tesseract', ruta_img, '-', '-l', 'spa'], check=True, capture_output=True, encoding='utf-8', errors='replace' ).stdout resultado.append((i, texto)) return resultado def intentar_texto_nativo(ruta_pdf): """Si el PDF tiene capa de texto real, es mucho más rápido que el OCR.""" try: salida = subprocess.run( ['pdftotext', '-layout', ruta_pdf, '-'], check=True, capture_output=True, encoding='utf-8', errors='replace' ).stdout # No basta con que haya "algo" de texto — muchas escrituras # escaneadas traen un sello/certificado con texto real en cada # página (poco contenido, pero se suma en documentos largos). Se # exige encontrar al menos una cláusula real para confiar en esto; # si no, es una señal de que el cuerpo del documento es imagen y # hay que ir a OCR. if PATRON_CLAUSULA.search(salida): paginas = salida.split('\x0c') # separador de página de pdftotext return [(i + 1, p) for i, p in enumerate(paginas) if p.strip()] except Exception: pass return None def detectar_clausulas(paginas_texto): """Devuelve lista de {numero, titulo, pagina, inicio, fin} por cláusula.""" texto_completo = '' mapa_offset_pagina = [] # (offset_inicio, num_pagina) for num_pag, texto in paginas_texto: mapa_offset_pagina.append((len(texto_completo), num_pag)) texto_completo += (texto or '') + '\n' def pagina_de(offset): pagina = paginas_texto[0][0] if paginas_texto else 1 for ini, num in mapa_offset_pagina: if ini <= offset: pagina = num else: break return pagina matches = list(PATRON_CLAUSULA.finditer(texto_completo)) clausulas = [] for i, m in enumerate(matches): ordinal = m.group(1).upper() numero = ORDINALES.get(ordinal, None) inicio = m.start() fin = matches[i + 1].start() if i + 1 < len(matches) else len(texto_completo) resto_linea = texto_completo[m.end():fin].split('\n', 1)[0].strip() clausulas.append({ 'numero': numero, 'ordinal_texto': ordinal, 'titulo': resto_linea[:80], 'pagina': pagina_de(inicio), 'inicio': inicio, 'fin': fin, }) return clausulas, texto_completo, pagina_de def buscar_requisitos(clausulas, texto_completo, pagina_de): resultado = {} for nombre_req, patrones in REQUISITOS.items(): encontrado = None for patron in patrones: m = re.search(patron, texto_completo, re.IGNORECASE | re.DOTALL) if m: offset = m.start() clausula_match = next( (c for c in clausulas if c['inicio'] <= offset < c['fin']), None ) encontrado = { 'clausula': clausula_match['numero'] if clausula_match else None, 'clausula_titulo': clausula_match['titulo'] if clausula_match else None, 'pagina': pagina_de(offset), } break resultado[nombre_req] = encontrado return resultado def detectar_programa(texto_completo): for programa, patrones in PATRON_PROGRAMA.items(): for patron in patrones: if re.search(patron, texto_completo, re.IGNORECASE): return programa return None def extraer_comprador(texto_completo): """ Busca en la sección de comparecencia (inicio del documento) el nombre y RUT de quien está identificado como "el comprador"/"la parte compradora"/"el deudor"/"el mutuario" — no cualquier persona nombrada (para no confundir con el representante del banco, de la inmobiliaria, o el notario). Es una heurística — el resultado siempre debe confirmarse a mano. """ inicio = texto_completo[:5000] # la comparecencia siempre está al principio etiquetas_comprador = re.compile( r'(el\s+comprador|la\s+parte\s+compradora|el\s+deudor|la\s+parte\s+deudora|el\s+mutuario)', re.IGNORECASE ) etiquetas_excluir = re.compile( r'(el\s+banco|el\s+vendedor|la\s+parte\s+vendedora|notari[oa]|en\s+representaci[oó]n\s+del?\s+banco)', re.IGNORECASE ) patron_persona = re.compile( r'(don|doña)\s+([A-ZÁÉÍÓÚÑ][A-ZÁÉÍÓÚÑ\s]+?),[\s\S]{0,300}?(?:c[eé]dula[^0-9]{0,40}|rut[^0-9]{0,10})(\d{1,2}\.?\d{3}\.?\d{3}[-‐]?[\dkK])', re.IGNORECASE ) candidatos = [] for m in patron_persona.finditer(inicio): # Ventana de contexto: desde este candidato hasta el próximo "don/doña" # (o el final de la comparecencia), para ver cómo lo etiquetan. ventana = inicio[m.end():m.end() + 400] es_comprador = etiquetas_comprador.search(ventana) es_excluido = etiquetas_excluir.search(ventana) if es_comprador and not es_excluido: candidatos.append({'nombre': m.group(2).strip(), 'rut': m.group(3)}) return candidatos def main(): if len(sys.argv) < 3: print(json.dumps({'error': 'Uso: extraer_escritura.py '})) sys.exit(1) ruta_pdf, ruta_salida = sys.argv[1], sys.argv[2] paginas_texto = intentar_texto_nativo(ruta_pdf) metodo = 'texto_nativo' if not paginas_texto: paginas_texto = ocr_pdf(ruta_pdf) metodo = 'ocr' clausulas, texto_completo, pagina_de = detectar_clausulas(paginas_texto) requisitos = buscar_requisitos(clausulas, texto_completo, pagina_de) programa = detectar_programa(texto_completo) candidatos_comprador = extraer_comprador(texto_completo) resultado = { 'metodo_lectura': metodo, 'total_paginas': len(paginas_texto), 'total_clausulas_detectadas': len(clausulas), 'programa_detectado': programa, 'candidatos_comprador': candidatos_comprador, 'requisitos': requisitos, } with open(ruta_salida, 'w', encoding='utf-8') as f: json.dump(resultado, f, ensure_ascii=False, indent=2) print(json.dumps({'ok': True, 'resumen': { 'clausulas': len(clausulas), 'programa': programa, 'compradores_candidatos': len(candidatos_comprador), }})) if __name__ == '__main__': main()
